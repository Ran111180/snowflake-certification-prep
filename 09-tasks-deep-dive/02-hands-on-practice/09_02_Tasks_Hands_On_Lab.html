9.2 Tasks Hands-On Lab

Objective: Build an enterprise-grade scheduled report refresh system with RBAC, stored procedures, and HTML email notifications.

Date: August 13, 2026

What you'll build:
SECURITYADMIN creates roles → SYSADMIN creates infrastructure → DATA_ENGINEER creates base tables, reports stored procedure, and scheduled task → Task refreshes 6 report tables biweekly and sends styled HTML email with row counts.

Architecture

+------------------+     +-------------------+     +------------------+
|  BASE Schema     | --> | Stored Procedure  | --> |  REPORTS Schema  |
|  (Source Tables) |     | REFRESH_ALL()     |     |  (Report Tables) |
+------------------+     +-------------------+     +------------------+
        |                         |                         |
        |                         v                         |
        |                 +---------------+                 |
        |                 | SEND_EMAIL()  |                 |
        |                 | (HTML report) |                 |
        |                 +---------------+                 |
        v                         ^                         v
+------------------+     +-------------------+     +------------------+
|  TASK (CRON)     | --> |  Executes every   | --> |  Email to inbox  |
|  Biweekly sched  |     |  1st & 15th       |     |  with row counts |
+------------------+     +-------------------+     +------------------+

0 Enterprise RBAC Setup

Role responsibilities:
SECURITYADMIN = creates roles & grants
SYSADMIN = creates databases & warehouses
ACCOUNTADMIN = account-level privileges only
DATA_ENGINEER = owns pipelines (our working role) 
-- ============================================================
-- STEP 0A: Create roles (SECURITYADMIN)
-- ============================================================
USE ROLE SECURITYADMIN;

CREATE ROLE IF NOT EXISTS DATA_ENGINEER
  COMMENT = 'Builds pipelines, creates tasks, owns transformations';

CREATE ROLE IF NOT EXISTS ANALYST
  COMMENT = 'Read-only access to report tables';

-- Role hierarchy (both roll up to SYSADMIN)
GRANT ROLE DATA_ENGINEER TO ROLE SYSADMIN;
GRANT ROLE ANALYST TO ROLE SYSADMIN;

-- Assign to your user
GRANT ROLE DATA_ENGINEER TO USER RANGAK565;
GRANT ROLE ANALYST TO USER RANGAK565;

-- ============================================================
-- STEP 0B: Create infrastructure (SYSADMIN)
-- ============================================================
USE ROLE SYSADMIN;

CREATE WAREHOUSE IF NOT EXISTS TASK_WH
  WAREHOUSE_SIZE = 'XSMALL'
  AUTO_SUSPEND = 60
  AUTO_RESUME = TRUE
  COMMENT = 'Warehouse for scheduled task execution';

CREATE DATABASE IF NOT EXISTS TASK_PRACTICE_DB
  COMMENT = 'Practice: Scheduled task refresh of report tables';

-- Grant ownership to DATA_ENGINEER
GRANT OWNERSHIP ON DATABASE TASK_PRACTICE_DB TO ROLE DATA_ENGINEER;
GRANT USAGE ON WAREHOUSE TASK_WH TO ROLE DATA_ENGINEER;
GRANT OPERATE ON WAREHOUSE TASK_WH TO ROLE DATA_ENGINEER;
GRANT USAGE ON WAREHOUSE TASK_WH TO ROLE ANALYST;

-- ============================================================
-- STEP 0C: Account-level privileges (ACCOUNTADMIN)
-- ============================================================
USE ROLE ACCOUNTADMIN;
GRANT EXECUTE TASK ON ACCOUNT TO ROLE DATA_ENGINEER;
GRANT EXECUTE MANAGED TASK ON ACCOUNT TO ROLE DATA_ENGINEER;

Verify Setup

SHOW GRANTS TO ROLE DATA_ENGINEER;
-- Should show: OWNERSHIP on TASK_PRACTICE_DB, USAGE+OPERATE on TASK_WH, EXECUTE TASK

3 Stored Procedure with HTML Email

What this does: Refreshes 6 report tables (CREATE OR REPLACE) then sends a styled HTML email with row counts for each table. 
CREATE OR REPLACE PROCEDURE REPORTS.REFRESH_ALL_REPORTS()
RETURNS STRING
LANGUAGE SQL
COMMENT = 'Refreshes all report tables and sends HTML email with row counts'
AS
DECLARE
  v_body STRING DEFAULT '';
  v_cnt1 INT; v_cnt2 INT; v_cnt3 INT;
  v_cnt4 INT; v_cnt5 INT; v_cnt6 INT;
BEGIN

  -- Report 1: Revenue by Region
  CREATE OR REPLACE TABLE REPORTS.RPT_REVENUE_BY_REGION AS
  SELECT a.region, COUNT(DISTINCT a.account_id) AS total_accounts,
    COUNT(o.opp_id) AS total_opportunities,
    SUM(CASE WHEN o.stage = 'Closed Won' THEN o.amount ELSE 0 END) AS won_revenue,
    SUM(o.amount) AS pipeline_total, CURRENT_TIMESTAMP() AS refreshed_at
  FROM BASE.ACCOUNTS a LEFT JOIN BASE.OPPORTUNITIES o ON a.account_id = o.account_id
  GROUP BY a.region;
  SELECT COUNT(*) INTO :v_cnt1 FROM REPORTS.RPT_REVENUE_BY_REGION;

  -- Report 2: Sales Rep Performance
  CREATE OR REPLACE TABLE REPORTS.RPT_SALES_REP_PERFORMANCE AS
  SELECT o.owner_name, COUNT(o.opp_id) AS total_deals,
    SUM(CASE WHEN o.stage = 'Closed Won' THEN 1 ELSE 0 END) AS deals_won,
    SUM(CASE WHEN o.stage = 'Closed Lost' THEN 1 ELSE 0 END) AS deals_lost,
    SUM(CASE WHEN o.stage = 'Closed Won' THEN o.amount ELSE 0 END) AS revenue_won,
    ROUND(SUM(CASE WHEN o.stage='Closed Won' THEN 1 ELSE 0 END)*100.0/COUNT(*),1) AS win_rate_pct,
    CURRENT_TIMESTAMP() AS refreshed_at
  FROM BASE.OPPORTUNITIES o GROUP BY o.owner_name;
  SELECT COUNT(*) INTO :v_cnt2 FROM REPORTS.RPT_SALES_REP_PERFORMANCE;

  -- Report 3: Pipeline by Stage
  CREATE OR REPLACE TABLE REPORTS.RPT_PIPELINE_BY_STAGE AS
  SELECT o.stage, COUNT(*) AS deal_count, SUM(o.amount) AS total_amount,
    AVG(o.amount) AS avg_deal_size, MIN(o.close_date) AS earliest_close,
    MAX(o.close_date) AS latest_close, CURRENT_TIMESTAMP() AS refreshed_at
  FROM BASE.OPPORTUNITIES o GROUP BY o.stage;
  SELECT COUNT(*) INTO :v_cnt3 FROM REPORTS.RPT_PIPELINE_BY_STAGE;

  -- Report 4: Account Health
  CREATE OR REPLACE TABLE REPORTS.RPT_ACCOUNT_HEALTH AS
  SELECT a.account_id, a.account_name, a.industry, a.region,
    COUNT(act.activity_id) AS total_activities,
    MAX(act.activity_date) AS last_activity_date,
    DATEDIFF('day', MAX(act.activity_date), CURRENT_DATE()) AS days_since_last_activity,
    CASE WHEN DATEDIFF('day', MAX(act.activity_date), CURRENT_DATE()) <= 30 THEN 'Healthy'
         WHEN DATEDIFF('day', MAX(act.activity_date), CURRENT_DATE()) <= 90 THEN 'At Risk'
         ELSE 'Inactive' END AS health_status,
    CURRENT_TIMESTAMP() AS refreshed_at
  FROM BASE.ACCOUNTS a LEFT JOIN BASE.ACTIVITIES act ON a.account_id = act.account_id
  WHERE a.is_active = TRUE GROUP BY a.account_id, a.account_name, a.industry, a.region;
  SELECT COUNT(*) INTO :v_cnt4 FROM REPORTS.RPT_ACCOUNT_HEALTH;

  -- Report 5: Industry Summary
  CREATE OR REPLACE TABLE REPORTS.RPT_INDUSTRY_SUMMARY AS
  SELECT a.industry, COUNT(DISTINCT a.account_id) AS account_count,
    SUM(a.annual_revenue) AS total_annual_revenue, COUNT(o.opp_id) AS opportunities,
    SUM(CASE WHEN o.stage = 'Closed Won' THEN o.amount ELSE 0 END) AS won_amount,
    CURRENT_TIMESTAMP() AS refreshed_at
  FROM BASE.ACCOUNTS a LEFT JOIN BASE.OPPORTUNITIES o ON a.account_id = o.account_id
  GROUP BY a.industry;
  SELECT COUNT(*) INTO :v_cnt5 FROM REPORTS.RPT_INDUSTRY_SUMMARY;

  -- Report 6: Monthly Activity
  CREATE OR REPLACE TABLE REPORTS.RPT_MONTHLY_ACTIVITY AS
  SELECT DATE_TRUNC('month', act.activity_date) AS activity_month,
    act.activity_type, COUNT(*) AS activity_count,
    SUM(act.duration_minutes) AS total_minutes,
    COUNT(DISTINCT act.account_id) AS accounts_touched,
    CURRENT_TIMESTAMP() AS refreshed_at
  FROM BASE.ACTIVITIES act
  GROUP BY DATE_TRUNC('month', act.activity_date), act.activity_type;
  SELECT COUNT(*) INTO :v_cnt6 FROM REPORTS.RPT_MONTHLY_ACTIVITY;

  -- Build HTML email body (see lab_queries.sql for full HTML template)
  v_body := '<html><body style="font-family:Arial">'
    || '<h2 style="color:#29B5E8">Report Refresh Complete</h2>'
    || '<table border="1" cellpadding="8">'
    || '<tr style="background:#29B5E8;color:white"><th>Report</th><th>Rows</th></tr>'
    || '<tr><td>RPT_REVENUE_BY_REGION</td><td>' || :v_cnt1 || '</td></tr>'
    || '<tr><td>RPT_SALES_REP_PERFORMANCE</td><td>' || :v_cnt2 || '</td></tr>'
    || '<tr><td>RPT_PIPELINE_BY_STAGE</td><td>' || :v_cnt3 || '</td></tr>'
    || '<tr><td>RPT_ACCOUNT_HEALTH</td><td>' || :v_cnt4 || '</td></tr>'
    || '<tr><td>RPT_INDUSTRY_SUMMARY</td><td>' || :v_cnt5 || '</td></tr>'
    || '<tr><td>RPT_MONTHLY_ACTIVITY</td><td>' || :v_cnt6 || '</td></tr>'
    || '</table><p>Total: ' || (:v_cnt1+:v_cnt2+:v_cnt3+:v_cnt4+:v_cnt5+:v_cnt6) || ' rows</p>'
    || '</body></html>';

  CALL SYSTEM$SEND_EMAIL(
    'LEARNING_EMAIL_NOTIFICATION',
    'ranga.k565@gmail.com',
    'Report Refresh - ' || CURRENT_DATE()::STRING,
    :v_body,
    'text/html'
  );

  RETURN '6 tables refreshed. Email sent. Total: ' || (:v_cnt1+:v_cnt2+:v_cnt3+:v_cnt4+:v_cnt5+:v_cnt6);
END;

4 Create & Schedule Task

-- Create the task (biweekly: 1st and 15th at midnight IST)
CREATE OR REPLACE TASK REPORTS.BIWEEKLY_REFRESH_TASK
  WAREHOUSE = TASK_WH
  SCHEDULE = 'USING CRON 0 0 1,15 * * Asia/Kolkata'
  COMMENT = 'Refreshes all 6 report tables biweekly'
AS
  CALL REPORTS.REFRESH_ALL_REPORTS();

-- Resume the task
ALTER TASK REPORTS.BIWEEKLY_REFRESH_TASK RESUME;

-- For testing: change to every 5 minutes
ALTER TASK REPORTS.BIWEEKLY_REFRESH_TASK SUSPEND;
ALTER TASK REPORTS.BIWEEKLY_REFRESH_TASK SET SCHEDULE = '5 MINUTE';
ALTER TASK REPORTS.BIWEEKLY_REFRESH_TASK RESUME;

-- Or test manually (works even when suspended)
CALL REPORTS.REFRESH_ALL_REPORTS();
