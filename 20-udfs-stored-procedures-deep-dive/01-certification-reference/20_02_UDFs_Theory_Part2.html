2. Stored Procedure Languages

SQL Scripting (Snowflake Scripting)
CREATE OR REPLACE PROCEDURE archive_old_orders(days_old NUMBER)
RETURNS VARCHAR
LANGUAGE SQL
EXECUTE AS OWNER
AS
$$
DECLARE
  row_count NUMBER;
  cutoff_date DATE;
BEGIN
  cutoff_date := DATEADD('day', -days_old, CURRENT_DATE());

  INSERT INTO orders_archive
  SELECT * FROM orders WHERE order_date < :cutoff_date;

  row_count := SQLROWCOUNT;

  DELETE FROM orders WHERE order_date < :cutoff_date;

  RETURN 'Archived ' || :row_count || ' orders';
EXCEPTION
  WHEN OTHER THEN
    ROLLBACK;
    RETURN 'Error: ' || SQLERRM;
END;
$$;

Key: Use :variable_name colon prefix to reference variables in SQL statements within the procedure body.

JavaScript Stored Procedure
CREATE OR REPLACE PROCEDURE load_data_with_retry(
  stage_path VARCHAR, target_table VARCHAR, max_retries NUMBER)
RETURNS VARCHAR
LANGUAGE JAVASCRIPT
EXECUTE AS CALLER
AS
$$
var attempts = 0;
var success = false;

while (attempts < MAX_RETRIES && !success) {
  try {
    var sql = `COPY INTO ${TARGET_TABLE}
               FROM @${STAGE_PATH}
               FILE_FORMAT = (TYPE = 'CSV')
               ON_ERROR = 'ABORT_STATEMENT'`;
    var stmt = snowflake.createStatement({sqlText: sql});
    var rs = stmt.execute();
    success = true;
    return "Success after " + (attempts + 1) + " attempt(s)";
  } catch(err) {
    attempts++;
    if (attempts >= MAX_RETRIES) {
      return "Failed after " + attempts + " attempts: " + err.message;
    }
  }
}
$$;

Key: Uses snowflake.createStatement() API. Parameters are UPPERCASED in JS body.

Python Stored Procedure
CREATE OR REPLACE PROCEDURE train_model(
  training_table VARCHAR, model_name VARCHAR)
RETURNS VARCHAR
LANGUAGE PYTHON
RUNTIME_VERSION = '3.11'
PACKAGES = ('snowflake-snowpark-python', 'scikit-learn', 'joblib')
HANDLER = 'run'
EXECUTE AS CALLER
AS
$$
import joblib
from sklearn.ensemble import RandomForestClassifier

def run(session, training_table, model_name):
    # Read training data
    df = session.table(training_table).to_pandas()

    X = df.drop(columns=['label'])
    y = df['label']

    # Train model
    model = RandomForestClassifier(n_estimators=100)
    model.fit(X, y)

    # Save model to stage
    model_path = f'/tmp/{model_name}.pkl'
    joblib.dump(model, model_path)
    session.file.put(model_path, f'@models/{model_name}.pkl', auto_compress=False)

    return f"Model {model_name} trained with accuracy: {model.score(X, y):.4f}"
$$;

Key: Python procedures receive a session object (Snowpark Session) as the first parameter automatically.

Java Stored Procedure
CREATE OR REPLACE PROCEDURE validate_email_batch(source_table VARCHAR)
RETURNS VARCHAR
LANGUAGE JAVA
RUNTIME_VERSION = '11'
PACKAGES = ('com.snowflake:snowpark:latest')
HANDLER = 'EmailValidator.run'
EXECUTE AS OWNER
AS
$$
import com.snowflake.snowpark_java.*;
import java.util.regex.Pattern;

public class EmailValidator {
  private static final Pattern EMAIL_PATTERN =
    Pattern.compile("^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+$");

  public static String run(Session session, String sourceTable) {
    DataFrame df = session.table(sourceTable);
    long totalRows = df.count();

    // Flag invalid emails
    session.sql(String.format(
      "UPDATE %s SET is_valid = FALSE WHERE NOT RLIKE(email, '^[A-Za-z0-9+_.-]+@[A-Za-z0-9.-]+$')",
      sourceTable
    )).collect();

    long invalidCount = session.sql(String.format(
      "SELECT COUNT(*) FROM %s WHERE is_valid = FALSE", sourceTable
    )).collect()[0].getLong(0);

    return String.format("Validated %d rows, %d invalid emails found", totalRows, invalidCount);
  }
}
$$;

3. EXECUTE AS OWNER vs CALLER

Execution Rights - Critical Exam Topic
EXECUTE AS OWNER (Default)
  • Procedure runs with owner's privileges
  • Caller does not need direct access to underlying objects
  • Owner must have privileges on all accessed objects
  • More secure: controlled access through the procedure
  • Cannot use caller's session variables/context
  • Cannot create objects owned by the caller
EXECUTE AS CALLER
  • Procedure runs with caller's privileges
  • Caller must have access to all underlying objects
  • Can use session context (current role, warehouse)
  • Objects created are owned by the caller
  • More flexible but less controlled
  • Required for procedures that manage caller's objects
-- Owner's rights: Callers can insert without direct table access
CREATE OR REPLACE PROCEDURE log_event(event_type VARCHAR, details VARCHAR)
RETURNS VARCHAR
LANGUAGE SQL
EXECUTE AS OWNER  -- callers don't need INSERT on audit_log
AS
$$
BEGIN
  INSERT INTO audit_log(event_time, event_type, details, caller_role)
  VALUES (CURRENT_TIMESTAMP(), :event_type, :details, CURRENT_ROLE());
  RETURN 'Event logged';
END;
$$;

-- Caller's rights: Operates on caller's objects
CREATE OR REPLACE PROCEDURE clone_schema(source_schema VARCHAR, target_schema VARCHAR)
RETURNS VARCHAR
LANGUAGE SQL
EXECUTE AS CALLER  -- needs caller's CREATE SCHEMA privilege
AS
$$
BEGIN
  CREATE SCHEMA IF NOT EXISTS IDENTIFIER(:target_schema)
    CLONE IDENTIFIER(:source_schema);
  RETURN 'Schema cloned: ' || :target_schema;
END;
$$;
Exam Tip: Security Implications

OWNER rights = controlled delegation (like a security gate). Callers can do specific actions without broad access.
CALLER rights = transparent execution. Procedure cannot do more than caller already can. Used for admin/utility procedures.

4. Anonymous Blocks (BEGIN...END)

Anonymous Blocks - Ad-hoc Procedural Logic

Anonymous blocks let you run procedural SQL without creating a stored procedure. They execute immediately and are not stored.

-- Simple anonymous block
BEGIN
  LET counter NUMBER := 0;
  LET target NUMBER := 5;

  FOR i IN 1 TO target DO
    counter := counter + i;
  END FOR;

  -- Return result
  RETURN counter;
END;

-- More complex: ETL with error handling
DECLARE
  tables_processed NUMBER DEFAULT 0;
  error_msg VARCHAR DEFAULT '';
BEGIN
  -- Process multiple tables
  FOR rec IN (SELECT table_name FROM information_schema.tables
              WHERE table_schema = 'RAW' AND table_type = 'BASE TABLE') DO
    BEGIN
      EXECUTE IMMEDIATE 'INSERT INTO staging.' || rec.table_name ||
                        ' SELECT * FROM raw.' || rec.table_name;
      tables_processed := tables_processed + 1;
    EXCEPTION
      WHEN OTHER THEN
        error_msg := error_msg || rec.table_name || ': ' || SQLERRM || '; ';
    END;
  END FOR;

  IF (error_msg != '') THEN
    RETURN 'Processed ' || tables_processed || ' tables. Errors: ' || error_msg;
  ELSE
    RETURN 'Successfully processed ' || tables_processed || ' tables';
  END IF;
END;
Anonymous Block Key Facts
  • No CALL needed - just execute the block directly
  • Cannot specify EXECUTE AS (always runs as caller)
  • Great for one-time admin tasks, migrations, testing
  • Support full Snowflake Scripting: IF, FOR, WHILE, LOOP, CURSOR, EXCEPTION
  • Can use DECLARE section before BEGIN or LET inside BEGIN

5. RESULTSET and Cursors

RESULTSET - Returning Query Results from Procedures
CREATE OR REPLACE PROCEDURE get_department_stats(dept_name VARCHAR)
RETURNS TABLE(metric VARCHAR, value NUMBER)
LANGUAGE SQL
AS
$$
DECLARE
  res RESULTSET;
BEGIN
  res := (SELECT 'headcount' AS metric, COUNT(*) AS value
          FROM employees WHERE department = :dept_name
          UNION ALL
          SELECT 'avg_salary', AVG(salary)
          FROM employees WHERE department = :dept_name
          UNION ALL
          SELECT 'total_budget', SUM(salary)
          FROM employees WHERE department = :dept_name);
  RETURN TABLE(res);
END;
$$;

-- Call it:
CALL get_department_stats('Engineering');
Cursors - Row-by-Row Processing
CREATE OR REPLACE PROCEDURE update_grades()
RETURNS VARCHAR
LANGUAGE SQL
AS
$$
DECLARE
  student_id NUMBER;
  avg_score FLOAT;
  new_grade VARCHAR;
  updated_count NUMBER DEFAULT 0;
  c1 CURSOR FOR
    SELECT id, average_score FROM students WHERE grade IS NULL;
BEGIN
  OPEN c1;

  FOR rec IN c1 DO
    student_id := rec.id;
    avg_score := rec.average_score;

    IF (avg_score >= 90) THEN
      new_grade := 'A';
    ELSEIF (avg_score >= 80) THEN
      new_grade := 'B';
    ELSEIF (avg_score >= 70) THEN
      new_grade := 'C';
    ELSE
      new_grade := 'F';
    END IF;

    UPDATE students SET grade = :new_grade WHERE id = :student_id;
    updated_count := updated_count + 1;
  END FOR;

  CLOSE c1;
  RETURN 'Updated ' || :updated_count || ' student grades';
END;
$$;
Cursor Key Points
  • Declared in DECLARE section with a query
  • Must OPEN before use (or use FOR rec IN cursor_name)
  • FOR loop auto-opens and auto-closes the cursor
  • FETCH INTO for manual iteration
  • Use sparingly - set-based operations are preferred for performance

6. Exception Handling

Snowflake Scripting Exception Handling
CREATE OR REPLACE PROCEDURE safe_division_demo(numerator NUMBER, denominator NUMBER)
RETURNS VARCHAR
LANGUAGE SQL
AS
$$
DECLARE
  result FLOAT;
  my_exception EXCEPTION (-20001, 'Custom: denominator cannot be zero');
BEGIN
  IF (denominator = 0) THEN
    RAISE my_exception;
  END IF;

  result := numerator / denominator;
  RETURN 'Result: ' || :result;

EXCEPTION
  WHEN my_exception THEN
    RETURN 'Error: ' || SQLERRM || ' (Code: ' || SQLCODE || ')';
  WHEN EXPRESSION_ERROR THEN
    RETURN 'Expression error: ' || SQLERRM;
  WHEN OTHER THEN
    RETURN 'Unexpected error: ' || SQLERRM;
END;
$$;
Built-in ExceptionWhen Raised
STATEMENT_ERRORSQL statement fails (syntax, permission, etc.)
EXPRESSION_ERRORExpression evaluation fails (divide by zero, overflow)
OTHERCatches any exception not caught by specific handlers

Exception Variables:

  • SQLCODE - Numeric error code
  • SQLERRM - Error message text
  • SQLSTATE - 5-character SQLSTATE code

7. External Functions

External Functions - Calling External APIs

External functions call remote services (AWS Lambda, Azure Functions, GCP Cloud Functions) via API integrations.

-- Step 1: Create API Integration (admin task)
CREATE OR REPLACE API INTEGRATION my_api_integration
  API_PROVIDER = aws_api_gateway
  API_AWS_ROLE_ARN = 'arn:aws:iam::123456789:role/snowflake-api-role'
  API_ALLOWED_PREFIXES = ('https://abc123.execute-api.us-east-1.amazonaws.com/')
  ENABLED = TRUE;

-- Step 2: Create External Function
CREATE OR REPLACE EXTERNAL FUNCTION translate_text(text VARCHAR, target_lang VARCHAR)
RETURNS VARIANT
API_INTEGRATION = my_api_integration
HEADERS = ('content-type' = 'application/json')
MAX_BATCH_ROWS = 100
AS 'https://abc123.execute-api.us-east-1.amazonaws.com/prod/translate';
PropertyDescription
API_INTEGRATIONReferences the API integration object (authentication)
MAX_BATCH_ROWSRows sent per batch to remote service (default 100)
COMPRESSIONAUTO, GZIP, DEFLATE, or NONE
HEADERSCustom HTTP headers sent with each request
CONTEXT_HEADERSSnowflake context (current_account, etc.) in headers
External Function Limitations
  • Always VOLATILE (results not cached)
  • Latency: network round-trip per batch
  • Cost: remote service charges apply
  • Batch processing: rows are batched (not one-at-a-time)
  • JSON format: input/output is JSON arrays
  • Cannot be SECURE (definition is already opaque)
  • Requires ACCOUNTADMIN for API_INTEGRATION setup

8. UDTF Advanced Patterns

Python UDTF with Partitioning
-- UDTF that computes running statistics per partition
CREATE OR REPLACE FUNCTION compute_partition_stats(value FLOAT)
RETURNS TABLE(
  min_val FLOAT, max_val FLOAT, avg_val FLOAT,
  stddev_val FLOAT, count_val NUMBER
)
LANGUAGE PYTHON
RUNTIME_VERSION = '3.11'
HANDLER = 'StatsComputer'
AS
$$
import statistics

class StatsComputer:
    def __init__(self):
        self._values = []

    def process(self, value):
        # Accumulate values within partition
        if value is not None:
            self._values.append(value)

    def end_partition(self):
        # Emit one row per partition with aggregate stats
        if self._values:
            yield (
                min(self._values),
                max(self._values),
                statistics.mean(self._values),
                statistics.stdev(self._values) if len(self._values) > 1 else 0.0,
                len(self._values)
            )
$$;

-- Usage with PARTITION BY:
SELECT category, stats.*
FROM sales_data,
     TABLE(compute_partition_stats(amount)
       OVER (PARTITION BY category)) stats;

UDTF Handler Methods:

MethodRequiredPurpose
__init__(self)OptionalInitialize state per partition
process(self, ...)RequiredCalled once per input row; can yield 0+ output rows
end_partition(self)OptionalCalled after all rows in partition; emit final rows

9. Snowpark UDFs & Deployment

Registering UDFs via Snowpark Python
# In a Snowpark session (notebook or script):
from snowflake.snowpark.functions import udf, udtf
from snowflake.snowpark.types import *
import pandas as pd

# Register a scalar UDF
@udf(
    name="my_db.my_schema.normalize_text",
    is_permanent=True,
    stage_location="@my_stage",
    replace=True,
    packages=["unidecode"]
)
def normalize_text(text: str) -> str:
    from unidecode import unidecode
    if text is None:
        return None
    return unidecode(text).lower().strip()

# Register a vectorized UDF (batch processing with pandas)
@udf(
    name="my_db.my_schema.batch_upper",
    is_permanent=True,
    stage_location="@my_stage",
    replace=True,
    input_types=[StringType()],
    return_type=StringType()
)
def batch_upper(series: pd.Series) -> pd.Series:
    return series.str.upper()

# Register via session (alternative approach)
session.udf.register(
    func=normalize_text,
    name="normalize_text_v2",
    stage_location="@my_stage",
    is_permanent=True,
    replace=True
)
Deploy with snow CLI

  • snow CLI: Build and deploy from project directory with snowflake.yml config

10. Production Patterns

Pattern 1: Stored Procedure as ETL Orchestrator
CREATE OR REPLACE PROCEDURE run_daily_etl(run_date DATE)
RETURNS TABLE(step VARCHAR, status VARCHAR, duration_seconds NUMBER)
LANGUAGE SQL
EXECUTE AS OWNER
AS
$$
DECLARE
  res RESULTSET;
  start_time TIMESTAMP;
  results ARRAY DEFAULT ARRAY_CONSTRUCT();
BEGIN
  -- Step 1: Load raw data
  start_time := CURRENT_TIMESTAMP();
  BEGIN
    EXECUTE IMMEDIATE 'COPY INTO raw_orders FROM @data_stage/orders/' ||
                      TO_CHAR(:run_date, 'YYYY/MM/DD/');
    results := ARRAY_APPEND(results,
      OBJECT_CONSTRUCT('step','load_raw','status','success',
        'duration', DATEDIFF('second', :start_time, CURRENT_TIMESTAMP())));
  EXCEPTION WHEN OTHER THEN
    results := ARRAY_APPEND(results,
      OBJECT_CONSTRUCT('step','load_raw','status','failed: '||SQLERRM,'duration',0));
    -- Continue to next step or abort based on severity
  END;

  -- Step 2: Transform
  start_time := CURRENT_TIMESTAMP();
  INSERT INTO staging_orders
  SELECT *, :run_date AS load_date FROM raw_orders WHERE load_date = :run_date;
  results := ARRAY_APPEND(results,
    OBJECT_CONSTRUCT('step','transform','status','success',
      'duration', DATEDIFF('second', :start_time, CURRENT_TIMESTAMP())));

  -- Step 3: Merge into final
  start_time := CURRENT_TIMESTAMP();
  MERGE INTO final_orders t USING staging_orders s
    ON t.order_id = s.order_id
    WHEN MATCHED THEN UPDATE SET t.status = s.status, t.updated_at = CURRENT_TIMESTAMP()
    WHEN NOT MATCHED THEN INSERT VALUES (s.order_id, s.customer_id, s.amount, s.status, CURRENT_TIMESTAMP());
  results := ARRAY_APPEND(results,
    OBJECT_CONSTRUCT('step','merge','status','success',
      'duration', DATEDIFF('second', :start_time, CURRENT_TIMESTAMP())));

  -- Return results as table
  res := (SELECT f.value:step::VARCHAR AS step,
                 f.value:status::VARCHAR AS status,
                 f.value:duration::NUMBER AS duration_seconds
          FROM TABLE(FLATTEN(:results)) f);
  RETURN TABLE(res);
END;
$$;
Pattern 2: Dynamic SQL with Parameterized Procedures
CREATE OR REPLACE PROCEDURE create_monthly_partition(
  base_table VARCHAR, year NUMBER, month NUMBER)
RETURNS VARCHAR
LANGUAGE SQL
EXECUTE AS CALLER
AS
$$
DECLARE
  partition_name VARCHAR;
  ddl_stmt VARCHAR;
BEGIN
  partition_name := :base_table || '_' || :year || '_' || LPAD(:month, 2, '0');

  ddl_stmt := 'CREATE TABLE IF NOT EXISTS ' || :partition_name || ' AS ' ||
              'SELECT * FROM ' || :base_table ||
              ' WHERE YEAR(created_at) = ' || :year ||
              ' AND MONTH(created_at) = ' || :month;

  EXECUTE IMMEDIATE :ddl_stmt;
  RETURN 'Created partition: ' || :partition_name;
END;
$$;
Pattern 3: Procedure with Logging and Audit
CREATE OR REPLACE PROCEDURE data_quality_check(schema_name VARCHAR)
RETURNS TABLE(table_name VARCHAR, check_type VARCHAR, result VARCHAR)
LANGUAGE SQL
EXECUTE AS OWNER
AS
$$
DECLARE
  res RESULTSET;
BEGIN
  -- Log start
  INSERT INTO procedure_audit_log(proc_name, start_time, parameters)
  VALUES ('data_quality_check', CURRENT_TIMESTAMP(),
          OBJECT_CONSTRUCT('schema', :schema_name));

  -- Run checks
  res := (
    SELECT table_name, 'null_check' AS check_type,
           CASE WHEN null_count > 0
                THEN 'FAIL: ' || null_count || ' nulls found'
                ELSE 'PASS' END AS result
    FROM (
      SELECT t.table_name, COUNT(*) AS null_count
      FROM information_schema.tables t
      JOIN information_schema.columns c
        ON t.table_name = c.table_name AND t.table_schema = c.table_schema
      WHERE t.table_schema = :schema_name
        AND c.is_nullable = 'NO'
      GROUP BY t.table_name
    )
  );

  -- Log completion
  UPDATE procedure_audit_log
  SET end_time = CURRENT_TIMESTAMP(), status = 'SUCCESS'
  WHERE proc_name = 'data_quality_check'
    AND end_time IS NULL;

  RETURN TABLE(res);
EXCEPTION
  WHEN OTHER THEN
    UPDATE procedure_audit_log
    SET end_time = CURRENT_TIMESTAMP(), status = 'FAILED', error_message = SQLERRM
    WHERE proc_name = 'data_quality_check' AND end_time IS NULL;
    RAISE;
END;
$$;
